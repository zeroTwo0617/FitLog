02  context = loadWhitelistContext(mode)

04  openid = cloud.serverIdentity()

              
05  if (!openid) return error("AUTH_REQUIRED")

              
06  prompt = buildPrompt(mode, query, context, session)

              
07  response = await LLM.chat(prompt, tools)

              
08  if (response.toolCall == "search_exercises")

              
09    matches = await searchExercises(response.args)

              
10    raw = await LLM.chat(prompt + matches, serverEnv)

              
11  parsed = parseJSON(raw)

              
12  result = validateSchema(parsed)

              
13  if (!result.valid) return textOnly(result.reply)

              
14  sessionId = await saveSession(openid, result.reply)

              
15  return { message: result.reply, draft: result.planDraft }
